library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity alu is
    Port (
        a   : in signed(3 downto 0);
        b   : in signed(3 downto 0);
        sel : in unsigned(2 downto 0);
        c   : out signed(3 downto 0)
    );
end alu;

architecture Behavioral of alu is

begin

    process(a, b, sel)
    begin
        case sel is

            when "000" =>
                c <= a and b;

            when "001" =>
                c <= a or b;

            when "010" =>
                c <= a nand b;

            when "011" =>
                c <= a nor b;

            when "100" =>
                c <= a;

            when "101" =>
                c <= a xnor b;

            when "110" =>
                c <= a + b;

            when "111" =>
                c <= a - b;

            when others =>
                c <= (others => '0');

        end case;
    end process;

end Behavioral;

Exp2: Shift Register
library IEEE;
use IEEE.STD_LOGIC_1164.ALL;

entity shift_register is
    Port (
        clk  : in STD_LOGIC;
        load : in STD_LOGIC;
        rst  : in STD_LOGIC;
        s    : in STD_LOGIC_VECTOR(1 downto 0);
        din  : in STD_LOGIC_VECTOR(3 downto 0);
        q    : out STD_LOGIC_VECTOR(3 downto 0)
    );
end shift_register;

architecture Behavioral of shift_register is

    signal qt : STD_LOGIC_VECTOR(3 downto 0);

begin

    process(clk, rst)
    begin

        if (rst = '1') then
            qt <= "0000";

        elsif (rising_edge(clk)) then

            qt <= din;

            case s is

                when "00" =>
                    qt(0) <= din(0);
                    qt(3 downto 1) <= qt(2 downto 0);

                when "01" =>
                    qt(0) <= din(0);
                    qt(3 downto 1) <= qt(2 downto 0);
                    qt <= din;

                when "10" =>
                    if (load = '1') then
                        qt <= din;
                    else
                        qt(3 downto 1) <= qt(2 downto 0);
                        qt(0) <= '0';
                    end if;

                when "11" =>
                    qt <= din;

                when others =>
                    qt <= "0000";

            end case;

        end if;

    end process;

    q <= qt;

end Behavioral;

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_ARITH.ALL;
use IEEE.STD_LOGIC_UNSIGNED.ALL;

entity updown_counter is
    Port (
        clk   : in STD_LOGIC;
        rst   : in STD_LOGIC;
        mode  : in STD_LOGIC;
        count : out STD_LOGIC_VECTOR(2 downto 0)
    );
end updown_counter;

architecture Behavioral of updown_counter is

    signal temp : STD_LOGIC_VECTOR(2 downto 0) := "000";

begin

    process(clk, rst)
    begin

        if (rst = '1') then
            temp <= "000";

        elsif (rising_edge(clk)) then

            if (mode = '0') then
                temp <= temp + 1;
            else
                temp <= temp - 1;
            end if;

        end if;

    end process;

    count <= temp;

end Behavioral;

Exp4:FIFO Memory

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.NUMERIC_STD.ALL;

entity fifo is
    generic (
        bits  : integer := 8;
        words : integer := 8
    );

    Port (
        clk      : in STD_LOGIC;
        wr_ena   : in STD_LOGIC;
        addr     : in integer range 0 to words-1;
        data_in  : in STD_LOGIC_VECTOR(bits-1 downto 0);
        data_out : out STD_LOGIC_VECTOR(bits-1 downto 0)
    );
end fifo;

architecture Behavioral of fifo is

    type vector_array is array (0 to words-1)
        of STD_LOGIC_VECTOR(bits-1 downto 0);

    signal memory : vector_array;

begin

    process(clk)
    begin
        if rising_edge(clk) then
            if wr_ena = '1' then
                memory(addr) <= data_in;
            end if;
        end if;
    end process;

    data_out <= memory(addr);

end Behavioral;

        data_in <= "10101010";
        wait until falling_edge(clk);

        data_in <= "11001100";
        wait until falling_edge(clk);

        -- Write data into memory location 2

        data_in <= "11110000";
        wait until falling_edge(clk);

        addr <= 0;
        wait for 10 ns;

        addr <= 1;
        wait for 10 ns;

Exp5:Keypad Interface

library IEEE;
use IEEE.STD_LOGIC_1164.ALL;
use IEEE.STD_LOGIC_ARITH.ALL;
use IEEE.STD_LOGIC_UNSIGNED.ALL;

entity keypad_interface is
    Port (
        clk         : in STD_LOGIC;
        reset       : in STD_LOGIC;
        rows        : in STD_LOGIC_VECTOR(3 downto 0);
        columns     : out STD_LOGIC_VECTOR(3 downto 0);
        key_press   : out STD_LOGIC;
        seg_en      : out STD_LOGIC_VECTOR(3 downto 0);
        seg_display : out STD_LOGIC_VECTOR(7 downto 0)
    );
end keypad_interface;

architecture Behavioral of keypad_interface is

    signal scan_clk      : STD_LOGIC;
    signal debounce_clk  : STD_LOGIC;
    signal clk_div       : STD_LOGIC_VECTOR(14 downto 0);
    signal col_signal    : STD_LOGIC_VECTOR(3 downto 0);
    signal D             : STD_LOGIC_VECTOR(5 downto 0);
    signal key_in        : STD_LOGIC;
    signal key_press_sig : STD_LOGIC;
    signal key_code_s    : STD_LOGIC_VECTOR(7 downto 0);

begin

    seg_en <= "0001";
    key_press <= key_press_sig;
    columns <= col_signal;

    scan_clk <= clk_div(14);
    debounce_clk <= clk_div(8);

    -- Clock divider
    process(clk, reset)
    begin
        if reset = '1' then
            clk_div <= (others => '0');
        else
            clk_div <= clk_div + 1;
        end if;
    end process;

    -- Scan pattern generator
    process(scan_clk, reset)
    begin
        if reset = '1' then
            col_signal <= "1110";
        elsif (scan_clk'event and scan_clk = '0') then
            col_signal <= col_signal(0) & col_signal(3 downto 1);
        end if;
    end process;

    -- Key detection
    key_in <= '1' when
        rows(0) = '0' or rows(1) = '0' or
        rows(2) = '0' or rows(3) = '0'
        else '0';

    -- Debouncing process
    process(debounce_clk, reset)
    begin
        if reset = '1' then
            D <= (others => '0');
        elsif (debounce_clk'event and debounce_clk = '0') then
            D <= key_in & D(5 downto 1);
        end if;
    end process;

    -- Key press detection
    key_press_sig <= D(0) and D(1) and D(2) and
                     D(3) and D(4) and D(5);

    -- Latching valid key
    process(key_press_sig, rows, reset)
    begin
        if reset = '1' then
            key_code_s <= (others => '0');
        elsif (key_press_sig = '1' and key_press_sig'event) then
            key_code_s <= col_signal & rows;
        end if;
    end process;

    -- Seven-segment display decoder
    seg_display <=
        "00000011" when key_code_s = "11101110" else
        "10011111" when key_code_s = "11101101" else
        "00100101" when key_code_s = "11101011" else
        "00001101" when key_code_s = "11100111" else
        "10011001" when key_code_s = "11011110" else
        "01001001" when key_code_s = "11011101" else
        "01000001" when key_code_s = "11011011" else
        "00011111" when key_code_s = "11010111" else
        "00000001" when key_code_s = "10111110" else
        "00001001" when key_code_s = "10111101" else
        "00000101" when key_code_s = "10111011" else
        "11000001" when key_code_s = "10110111" else
        "01100011" when key_code_s = "01111110" else
        "10000101" when key_code_s = "01111101" else
        "01100001" when key_code_s = "01111011" else
        "01110001" when key_code_s = "01110111" else
        "00000000";

end Behavioral;

            seg_display => seg_display
        );

    begin
        clk <= '0';
        wait for 5 ns;
        clk <= '1';
        wait for 5 ns;
    end process;

    begin
        -- Apply reset
        reset <= '1';
        wait for 100 ns;

        reset <= '0';
        rows <= "1111";
        wait for 10 us;

        -- Press a keypad button
        rows <= "1110";
        wait for 100 us;

        -- Release the button
        rows <= "1111";
        wait for 50 us;

        wait;
    end process;
